cmake_minimum_required(VERSION 3.22)
project(SwellScaleFinder VERSION 0.1.1)
set(CMAKE_CXX_STANDARD 17)
if(MSVC)
    add_compile_options(/utf-8)
endif()
include(FetchContent)
FetchContent_Declare(JUCE GIT_REPOSITORY https://github.com/juce-framework/JUCE.git GIT_TAG 8.0.4 GIT_SHALLOW TRUE)
FetchContent_MakeAvailable(JUCE)
set(formats VST3 Standalone)
if(APPLE)
    list(APPEND formats AU)
endif()
juce_add_plugin(ScaleFinder COMPANY_NAME "Swell Audio" BUNDLE_ID fr.swell-audio.scalefinder
 ICON_BIG scale_finder.png
 PLUGIN_MANUFACTURER_CODE Swel PLUGIN_CODE ScFi FORMATS ${formats}
 PRODUCT_NAME "Swell Scale Finder" NEEDS_MIDI_INPUT TRUE EDITOR_WANTS_KEYBOARD_FOCUS TRUE)
juce_add_binary_data(Brand SOURCES scale_finder.png)
target_sources(ScaleFinder PRIVATE Source/Finder.cpp)
target_compile_definitions(ScaleFinder PUBLIC JUCE_WEB_BROWSER=0 JUCE_USE_CURL=0 JUCE_VST3_CAN_REPLACE_VST2=0)
target_link_libraries(ScaleFinder PRIVATE Brand juce::juce_audio_utils)
option(SWELL_TESTS "Build functional checks" OFF)
if(SWELL_TESTS)
    add_executable(ScaleCheck Tests/Check.cpp)
    target_link_libraries(ScaleCheck PRIVATE ScaleFinder)
    target_include_directories(ScaleCheck PRIVATE $<TARGET_PROPERTY:ScaleFinder,INCLUDE_DIRECTORIES>)
    target_compile_definitions(ScaleCheck PRIVATE $<TARGET_PROPERTY:ScaleFinder,COMPILE_DEFINITIONS>)
endif()
